-- 2. TÁBLÁK LÉTREHOZÁSA

-- Felhasználók tábla
CREATE TABLE Users (
    id int IDENTITY(1,1) PRIMARY KEY,
    nev nvarchar(100) NOT NULL,
    email nvarchar(100) UNIQUE NOT NULL,
    password nvarchar(255) NOT NULL,
    role nvarchar(20) DEFAULT 'user'
);

-- Túrák tábla
CREATE TABLE Turak (
    id int IDENTITY(1,1) PRIMARY KEY,
    nev nvarchar(255) NOT NULL,
    helyszin nvarchar(255) NOT NULL,
    idotartam nvarchar(50) NOT NULL,
    nehezseg nvarchar(50) NOT NULL,
    leiras nvarchar(MAX) NULL,
    kinek_ajanljuk nvarchar(MAX) NULL
);

-- Fórum Témák tábla 
CREATE TABLE ForumTemak (
    id int IDENTITY(1,1) PRIMARY KEY,
    cim nvarchar(255) NOT NULL,
    user_id int NOT NULL, 
    datum datetime DEFAULT GETDATE(),
    CONSTRAINT FK_ForumTemak_Users FOREIGN KEY (user_id) REFERENCES Users(id) ON DELETE CASCADE
);

-- Fórum Kommentek tábla 
CREATE TABLE ForumKommentek (
    id int IDENTITY(1,1) PRIMARY KEY,
    tema_id int NOT NULL,
    user_id int NOT NULL, 
    szoveg nvarchar(MAX) NOT NULL,
    datum datetime DEFAULT GETDATE(),
    CONSTRAINT FK_Komment_Tema FOREIGN KEY (tema_id) REFERENCES ForumTemak(id) ON DELETE CASCADE,
    CONSTRAINT FK_Komment_User FOREIGN KEY (user_id) REFERENCES Users(id)
);

-- Fórum Téma Lájk tábla (Kapcsolótábla a szívezéshez)
CREATE TABLE TemaLikes (
    id INT IDENTITY(1,1) PRIMARY KEY,
    tema_id INT NOT NULL,
    user_id INT NOT NULL,
    datum DATETIME DEFAULT GETDATE(),
    CONSTRAINT FK_Like_Tema FOREIGN KEY (tema_id) REFERENCES ForumTemak(id) ON DELETE CASCADE,
    CONSTRAINT FK_Like_User FOREIGN KEY (user_id) REFERENCES Users(id),
    CONSTRAINT UC_UserTemaLike UNIQUE (tema_id, user_id)
);

-- Kedvenc Túrák tábla
CREATE TABLE KedvencTurak (
    id INT IDENTITY(1,1) PRIMARY KEY,
    user_id INT NOT NULL,
    tura_id INT NOT NULL,
    datum DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (user_id) REFERENCES Users(id) ON DELETE CASCADE,
    FOREIGN KEY (tura_id) REFERENCES Turak(id) ON DELETE CASCADE,
    CONSTRAINT UC_UserTura UNIQUE (user_id, tura_id)
);
GO
